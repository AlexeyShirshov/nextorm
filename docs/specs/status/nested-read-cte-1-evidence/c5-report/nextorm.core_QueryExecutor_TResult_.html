 762    /// <summary>

 766    /// </summary>
 767    private static TResult ConvertScalar<TResult>(object value)
 1768    {
 769        // Already the wanted runtime type (bool/int/long/double/decimal/string/...): unbox, no
 770        // IConvertible dispatch. A boxed underlying value also satisfies `is T?`.
 1771        if (value is TResult result) return result;
 772
 773        // A nullable projection (TResult = T?) is converted through its underlying type:
 774        // Convert.ChangeType does not understand Nullable<T>, and the typed fast paths below compare
 775        // against the non-nullable type. Boxing the underlying value is a valid unboxing to T?.
