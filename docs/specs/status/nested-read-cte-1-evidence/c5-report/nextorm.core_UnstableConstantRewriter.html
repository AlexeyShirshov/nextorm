 640    {

 642        // must be a compile-time constant and is rendered as a one-character string.

 648            && node.Arguments is [var charArg, var countArg])
 649        {

 651                throw new NotSupportedException("The new string(char, n) character must be a constant.");

 655                Visit(charArg);
 656                Visit(countArg);

 658            }

 662                SqlLiteral.ToSqlStringLiteral(character),
 663                VisitToString(countArg)));

 665        }

 667        if (TupleSqlTranslator.TryTranslateNew(this, node))
 668            return node;
 669
 670        // A value-type constructor that does not reference the query (for example
 671        // new DateTime(2014, 3, 20)) is a constant. Falling through to base.VisitNew would visit the
 672        // constructor arguments and concatenate their literals into meaningless SQL ("2014320"), so

 714    /// <summary>
 715    /// Appends a constant literal (null, string/GUID, or primitive) to the builder. Returns
 716    /// <see langword="false"/> for values the caller must handle itself (anything non-primitive).
 717    /// </summary>
 718    private bool TryEmitValue(Type t, object? v)
 719    {
 720        if (v is null)
 721            _builder!.Append(Kw("null"));
 722        else if (t == typeof(string) || t == typeof(Guid))
 723        {
 724            _builder!.Append('\'').Append(v.ToString()).Append('\'');
 725        }
 726        else if (t.IsPrimitive || t == typeof(decimal))
 727        {
 728            if (t == typeof(bool))
 729                _builder!.Append(_dialect.MakeBool((bool)v));
 730            else
 731                // SQL literals must not depend on the current culture (ru-RU would render 0.2
 732                // as 0,2, which changes the statement). decimal is not an IsPrimitive type, so it
 733                // has to be listed explicitly or its literal would be dropped from the SQL.
 734                _builder!.Append(Convert.ToString(v, CultureInfo.InvariantCulture));
 735        }
 736        else if (t == typeof(TimeSpan) && v is TimeSpan duration)
 737        {
 738            // A duration literal: on a provider without a native type it is the stored integer in the
 739            // current duration context; a native provider (or a context-free duration) binds it as a
 740            // parameter so the driver maps it to its native type.
 741            if (DurationUnitContext is { } unit)
 742                _builder!.Append(DurationStorage.ToStorage(duration, unit).ToString(CultureInfo.InvariantCulture));
 743            else
 744            {
 745                var paramName = _parameterProvider.GetParamName();
 746                _params.Add(new Parameter(paramName, duration));
 747                _builder!.Append(_dialect.MakeParam(paramName));
 748            }
 749        }
 750        else
 751            return false;
 752
 753        return true;
 754    }
