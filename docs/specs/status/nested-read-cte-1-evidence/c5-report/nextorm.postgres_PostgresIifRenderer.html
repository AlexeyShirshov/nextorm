 678    private static bool TryMapDate(string format, out string mask)
 679    {
 680        var sb = new StringBuilder(format.Length + 4);
 681        var i = 0;
 682
 683        while (i < format.Length)
 684        {
 685            if (Match(format, i, "yyyy")) { sb.Append("YYYY"); i += 4; }
 686            else if (Match(format, i, "yy")) { sb.Append("YY"); i += 2; }
 687            else if (Match(format, i, "MM")) { sb.Append("MM"); i += 2; }
 688            else if (Match(format, i, "dd")) { sb.Append("DD"); i += 2; }
 689            else if (Match(format, i, "HH")) { sb.Append("HH24"); i += 2; }
 690            else if (Match(format, i, "mm")) { sb.Append("MI"); i += 2; }
 691            else if (Match(format, i, "ss")) { sb.Append("SS"); i += 2; }
 692            else if (format[i] is '-' or '/' or '.' or ':' or ' ') { sb.Append(format[i]); i++; }
 693            else { mask = string.Empty; return false; }
 694        }
 695
 696        mask = sb.ToString();
 697        return true;
 698    }
