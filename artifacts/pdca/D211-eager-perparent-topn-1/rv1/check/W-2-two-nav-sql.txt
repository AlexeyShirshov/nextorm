select t1.id, t1.name, t3.id, t3.ParentId, t3.SortKey, t3.flag, t5.id, t5.ParentId, t5.rank from edge_parent as 't1' left join (select t2.id, t2.parent_id as 'ParentId', t2.sort_key as 'SortKey', t2.flag
from (select *, row_number() over (partition by parent_id order by sort_key) as '__nextorm_rn' from edge_child) as 't2'
 where t2.__nextorm_rn <= 3) as 't3' on t1.id = t3.ParentId left join (select t4.id, t4.parent_id as 'ParentId', t4.rank
from (select *, row_number() over (partition by parent_id order by rank) as '__nextorm_rn' from edge_note) as 't4'
 where t4.__nextorm_rn <= 3) as 't5' on t1.id = t5.ParentId
 order by t1.id, t3.SortKey, t5.rank
