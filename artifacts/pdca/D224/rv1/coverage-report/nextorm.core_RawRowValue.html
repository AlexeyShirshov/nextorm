 391    /// <summary>
 392    /// Builds the accessor for a caller-registered named composite (#194, S5): the driver resolves the
 393    /// single column to the registered CLR type, so it is read through the typed generic accessor. This
 394    /// accessor lives in the raw path (not in the provider's <c>MapColumnExpression</c>) so an ordinary
 395    /// LINQ class column — which may carry a value converter — is never hijacked. The driver read is
 396    /// wrapped (<see cref="RawRowValue.ReadNamedComposite{T}"/>) so a read failure or a SQL NULL into a
 397    /// non-nullable struct composite surfaces the R194-ERROR contract instead of a bare driver exception.
 398    /// Whole-record <c>DBNull</c> is still taken first via <c>IsDBNull</c> for a nullable target.
 399    /// </summary>
 400    private static Expression MapNamedCompositeColumn(SelectExpression column, Expression param)
 401    {
 402        var realType = Nullable.GetUnderlyingType(column.PropertyType) ?? column.PropertyType;
 403        var index = Expression.Constant(column.Index);
 404        Expression getter = Expression.Call(
 405            ReadNamedCompositeMI.MakeGenericMethod(realType),
 406            Expression.Convert(param, typeof(DbDataReader)),
 407            index);
 408
 409        if (!column.Nullable)
 410            return getter;
 411
 412        // The getter returns the unwrapped composite type; for a declared non-nullable struct composite
 413        // that is the property type, but for a declared Nullable<struct> it must be lifted so the two
 414        // branches of the whole-record-NULL condition share a type.
 415        if (getter.Type != column.PropertyType)
 416            getter = Expression.Convert(getter, column.PropertyType);
 417
 418        return Expression.Condition(
 419            Expression.Call(param, IsDBNullMI, index),
 420            Expression.Constant(null, column.PropertyType),
 421            getter);

 424    /// <summary>
 425    /// Builds the accessor for an anonymous PostgreSQL <c>ROW(...)</c>: the driver already produced a
 426    /// null-preserving <c>object[]</c> for the single record column, so the mapper validates the declared
 427    /// tuple shape and constructs <see cref="Tuple"/> from it. A whole-record SQL NULL becomes a CLR
 428    /// <see langword="null"/>; per-field NULL/error semantics live in <see cref="RawRowValue"/>.
 429    /// </summary>
 430    private static Expression MapAnonymousTupleColumn(SelectExpression column, Expression record)
 431    {
 432        var tupleType = column.PropertyType;
 433        var args = tupleType.GetGenericArguments();
 434        foreach (var arg in args)
 435        {
 436            if (!IsScalarType(arg))
 437                throw UnsupportedRawRow("a record field type is outside the supported scalar allow-list, or a nested ROW
 438        }

 440        var index = Expression.Constant(column.Index);
 441        // Validate the driver's actual record at runtime: a non-array value (tuple materialization enabled)
 442        // or a ROW arity that differs from the declared tuple must not silently drop fields or throw a bare
 443        // IndexOutOfRangeException (R194-ERROR).
 444        var values = Expression.Call(
 445            ValidateRecordMI,
 446            Expression.Call(record, GetValueMI, index),
 447            Expression.Constant(args.Length));
 448
 449        var ctor = tupleType.GetConstructor(args)
 450            ?? throw UnsupportedRawRow($"{tupleType.Name} has no public constructor matching its arity");
 451
 452        var items = new Expression[args.Length];
 453        for (var i = 0; i < items.Length; i++)
 454        {
 455            items[i] = Expression.Call(
 456                ConvertFieldMI.MakeGenericMethod(args[i]),

 458                Expression.Constant(i));
 459        }

 461        return Expression.Condition(
 462            Expression.Call(record, IsDBNullMI, index),
 463            Expression.Constant(null, tupleType),
 464            Expression.New(ctor, items));
 465    }

 759/// R194-NULL / R194-ERROR contract: a SQL NULL for a non-nullable value-type item fails (never
 760/// <c>default(T)</c>), a nested ROW is an unsupported shape, and a conversion failure preserves its
 761/// inner exception.

 764{
